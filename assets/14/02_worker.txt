"""One recorded FD001 fold: load saved AE, refit selected classical detectors, compare scores."""
from pathlib import Path
import json,sys,hashlib,traceback,platform


def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()


def require(condition,message):
    if not bool(condition):raise ValueError(message)


import numpy as np
import pandas as pd
from sklearn.ensemble import IsolationForest
from sklearn.neighbors import LocalOutlierFactor,NearestNeighbors
def fit_candidate(method,params,X,seed):
    if method=='IF':
        model=IsolationForest(**params,max_samples='auto',contamination='auto',random_state=seed,n_jobs=-1)
    elif method=='LOF':
        require(params['n_neighbors']<len(X),'Insufficient fit rows for LOF')
        model=LocalOutlierFactor(**params,novelty=True,contamination='auto',metric='euclidean',n_jobs=-1)
    elif method=='kNN':
        require(params['n_neighbors']<=len(X),'Insufficient fit rows for kNN')
        model=NearestNeighbors(**params,metric='euclidean',n_jobs=-1)
    else:
        raise ValueError('Unsupported reproduction method: '+str(method))
    return model.fit(X)


def score(method,model,X):
    if method=='kNN':values=model.kneighbors(X,return_distance=True)[0][:,-1]
    elif method in ['IF','LOF']:values=-model.score_samples(X)
    else:raise ValueError('Unsupported scoring method: '+str(method))
    require(np.isfinite(values).all(),'Nonfinite scores')
    return values


# Explicit missing-method implementations; no generic fallback to kNN.
import time,warnings,gc
import joblib
from sklearn.cluster import DBSCAN
from sklearn.svm import OneClassSVM
from sklearn.neighbors import BallTree
from sklearn.exceptions import ConvergenceWarning


class CandidateInvalid(ValueError):
    pass


class ResourceLimit(RuntimeError):
    pass


def bounded_dbscan(F, eps, min_samples, entry_limit):
    """Exact unweighted Euclidean DBSCAN with bounded radius-neighbor batches."""
    from collections import deque
    cap = min(250_000, int(entry_limit))
    n = len(F)
    require(0 < n <= cap, 'A single neighborhood might exceed the entry budget')
    # Even a fully dense batch can return at most cap neighbor indices.
    width = min(256, cap // n)
    search = NearestNeighbors(radius=eps, algorithm='ball_tree', leaf_size=30,
                              metric='euclidean', n_jobs=1).fit(F)
    peak_entries = 0

    def neighborhoods(indices):
        nonlocal peak_entries
        found = search.radius_neighbors(F[indices], return_distance=False)
        entries = sum(len(neighbors) for neighbors in found)
        require(entries <= cap, 'Radius batch exceeded budget')
        peak_entries = max(peak_entries, entries)
        return found

    counts = np.empty(n, dtype=np.intp)
    for start in range(0, n, width):
        stop = min(start + width, n)
        counts[start:stop] = [len(row) for row in neighborhoods(slice(start, stop))]
    core = counts >= min_samples
    core_indices = np.flatnonzero(core)
    if not len(core_indices):
        raise CandidateInvalid('No fit-set core samples; core-distance score undefined')

    # Start components in native input order. Only core points expand a cluster.
    # An ambiguous border point belongs to the first component that reaches it.
    labels = np.full(n, -1, dtype=np.intp)
    cluster = 0
    for first in core_indices:
        if labels[first] != -1:
            continue
        labels[first] = cluster
        pending = deque([int(first)])
        while pending:
            batch = [pending.popleft() for _ in range(min(width, len(pending)))]
            for neighbors in neighborhoods(batch):
                fresh = neighbors[labels[neighbors] == -1]
                labels[fresh] = cluster
                pending.extend(fresh[core[fresh]].tolist())
        cluster += 1

    db = DBSCAN(eps=eps, min_samples=min_samples, metric='euclidean',
                algorithm='ball_tree', n_jobs=-1)
    db.core_sample_indices_ = core_indices
    db.components_ = F[core_indices].copy()
    db.labels_ = labels
    db.n_features_in_ = F.shape[1]
    diagnostics = {'backend':'bounded_radius_batches_v1', 'batch_rows':width,
                   'batch_entry_cap':cap, 'peak_batch_entries':peak_entries,
                   'counted_neighbor_entries':int(counts.sum())}
    return db, diagnostics


def verify_stream_backend():
    # Run once in the actual worker's sklearn version before using the fallback.
    if getattr(verify_stream_backend, 'passed', False):
        return
    probe = np.array([[-1.75]] * 4 + [[-1.0], [0.0], [1.0]] + [[1.75]] * 4 + [[4.0]])
    for values in (probe, probe[::-1].copy()):
        ref = DBSCAN(eps=1., min_samples=4, algorithm='ball_tree', metric='euclidean', n_jobs=1).fit(values)
        got, _ = bounded_dbscan(values, 1., 4, len(values))
        require(np.array_equal(ref.core_sample_indices_, got.core_sample_indices_) and
                np.array_equal(ref.labels_, got.labels_) and np.array_equal(ref.components_, got.components_),
                'Native/streamed DBSCAN compatibility check failed')
    verify_stream_backend.passed = True


def fit_missing(method,params,X,limits):
    F=np.ascontiguousarray(X,dtype=np.float64)
    require(np.isfinite(F).all(),'Nonfinite fitting features')
    if method=='DBSCAN':
        eps=float(params['eps']);m=int(params['min_samples'])
        if eps<=0:raise CandidateInvalid('Nonpositive eps from fit-only distance quantile')
        tree=BallTree(F,metric='euclidean')
        counts=tree.query_radius(F,r=eps,count_only=True)
        entries=int(counts.sum())
        if entries > limits['max_dbscan_neighbor_entries']:
            verify_stream_backend()
            db, backend_info = bounded_dbscan(F, eps, m, limits['max_dbscan_neighbor_entries'])
        else:
            db = DBSCAN(eps=eps,min_samples=m,metric='euclidean',algorithm='ball_tree',n_jobs=-1).fit(F)
            backend_info = {'backend':'native_sklearn'}
        if len(db.core_sample_indices_)==0:raise CandidateInvalid('No fit-set core samples; core-distance score undefined')
        core_nn=NearestNeighbors(n_neighbors=1,algorithm='ball_tree',metric='euclidean',n_jobs=-1).fit(db.components_)
        model={'kind':'DBSCAN_core_distance_v1','dbscan':db,'core_search':core_nn,'eps':eps}
        diag={'core_samples':int(len(db.core_sample_indices_)),'fit_rows':len(F),'fit_noise_fraction':float(np.mean(db.labels_==-1)),
              'clusters':int(len(set(db.labels_)-{-1})),'neighbor_entries':entries,'eps':eps,'min_samples':m}
        diag.update(backend_info)
        return model,diag
    if method=='OC-SVM':
        with warnings.catch_warnings(record=True) as recorded:
            warnings.simplefilter('always',ConvergenceWarning)
            model=OneClassSVM(kernel='rbf',nu=params['nu'],gamma=params['gamma'],tol=1e-3,
                              cache_size=512,shrinking=True,max_iter=100000).fit(F)
        if model.fit_status_!=0 or any(issubclass(w.category,ConvergenceWarning) for w in recorded):
            raise CandidateInvalid('OC-SVM did not converge within the frozen iteration limit')
        return model,{'support_vectors':int(len(model.support_)),'fit_rows':len(F),'fit_status':int(model.fit_status_),
                      'iterations':int(model.n_iter_),'effective_gamma':float(model._gamma)}
    raise ValueError('Unsupported missing method: '+str(method))


def score_missing(method,model,X):
    Q=np.ascontiguousarray(X,dtype=np.float64)
    if method=='DBSCAN':
        values=model['core_search'].kneighbors(Q,return_distance=True)[0][:,0]/model['eps']
    elif method=='OC-SVM':
        values=-model.score_samples(Q)
    else:raise ValueError('Unsupported missing scoring method: '+str(method))
    require(np.isfinite(values).all(),'Nonfinite new-method scores')
    return values


def prepare(config):
    import numpy as np
    import pandas as pd
    import tensorflow as tf
    from sklearn.preprocessing import StandardScaler
    from threadpoolctl import threadpool_limits

    root = Path(config['project'])
    revision = root / 'revision_outputs'
    for relative, expected in config['known_hashes'].items():
        require(digest(revision / relative) == expected, 'Kaynak değişmiş: ' + relative)
    train = root / 'data/train_FD001.txt'
    require(digest(train) == config['train_sha256'], 'FD001 eğitim dosyası değişmiş')
    folder = revision / config['fold_relative_directory']
    protocol = revision / '02_FD001_protocol'
    meta = pd.read_csv(protocol/'train_evaluation_reference.csv')
    roles = pd.read_csv(protocol/'engine_splits.csv')
    split = roles.loc[roles.scheme.eq('cv') & roles.fold.eq(1)].set_index('engine_key', verify_integrity=True)
    role = meta.engine_key.map(split.role)
    require(role.notna().all() and split.role.value_counts().to_dict()=={'fit':70,'early_stop':10,'selection':10,'outer_valid':10}, 'Motor bölümü farklı')
    raw = pd.read_csv(train, sep=r'\s+', header=None)
    require(raw.shape==(20631,26) and np.array_equal(raw.iloc[:,:2],meta[['UnitNumber','Time']]), 'Satır eşlemesi başarısız')
    sensors = raw.iloc[:,5:].to_numpy(dtype=np.float32)
    fit, outer = role.eq('fit'), role.eq('outer_valid')
    ref = meta.loc[outer,['sample_key','UnitNumber','Time','RUL','late_life_le_30']].copy()
    require(ref.sample_key.is_unique and np.array_equal(ref.late_life_le_30,ref.RUL.le(30)), 'Referans uyuşmuyor')
    stored = json.loads((folder/'scaler_parameters.json').read_text())
    require(stored['input']['features']==[f'Sensor{i}' for i in range(1,22)], 'Sensör sırası farklı')
    def scale(train_X, outer_X, name):
        scaler = StandardScaler().fit(train_X)
        for attr, key in [('mean_', 'mean'), ('scale_', 'scale')]:
            np.testing.assert_allclose(getattr(scaler,attr),stored[name][key],rtol=1e-5,atol=1e-7)
        return scaler.transform(train_X), scaler.transform(outer_X)
    Xfit, Xouter = scale(sensors[fit],sensors[outer],'input')
    tf.keras.utils.set_random_seed(42)
    tf.config.experimental.enable_op_determinism()
    encoder = tf.keras.models.load_model(folder/'encoder.keras',compile=False)
    zfit, zouter = [encoder.predict(x,batch_size=256,verbose=0) for x in [Xfit,Xouter]]
    Zfit, Zouter = scale(zfit,zouter,'latent')
    require(Xouter.shape==Zouter.shape==(len(ref),21),'Boyut uyuşmuyor')
    return Zfit, Zouter, meta.loc[fit,"sample_key"], ref.reset_index(drop=True)

def contributions(score_fn, backgrounds, point, seed=42, repeats=8):
    # Empirical marginal Shapley game. Each fitting background completes
    # the missing coordinates; average paired forward/reverse paths.
    rng = np.random.default_rng(seed)
    n, dim = backgrounds.shape
    responses = np.asarray(score_fn(backgrounds),dtype=float)
    baseline = float(responses.mean())
    endpoint = float(score_fn(point[None,:])[0])
    estimates = []
    for _ in range(repeats):
        order = np.stack([rng.permutation(dim) for _ in range(n)])
        paths = []
        orders = []
        for perm in (order,order[:,::-1]):
            path = np.repeat(backgrounds[:,None,:],dim+1,axis=1)
            for step in range(dim):
                path[:,step+1] = path[:,step]
                col = perm[:,step]
                path[np.arange(n),step+1,col] = point[col]
            paths.append(path)
            orders.append(perm)
        points = np.concatenate(paths).reshape(-1,dim)
        values = np.concatenate([np.asarray(score_fn(points[i:i+256]),dtype=float) for i in range(0,len(points),256)]).reshape(2*n,dim+1)
        np.testing.assert_allclose(values[:,0],np.tile(responses,2),rtol=1e-5,atol=1e-6)
        np.testing.assert_allclose(values[:,-1],endpoint,rtol=1e-5,atol=1e-6)
        aligned = np.empty((2*n,dim))
        np.put_along_axis(aligned,np.concatenate(orders),np.diff(values,axis=1),axis=1)
        estimates.append(aligned.mean(axis=0))
    phi=np.mean(estimates,axis=0)
    np.testing.assert_allclose(baseline+phi.sum(),endpoint,rtol=1e-5,atol=2e-6)
    return dict(base_score=baseline,score=endpoint,phi=phi.tolist(),
                permutation_se=(np.std(estimates,axis=0,ddof=1)/np.sqrt(repeats)).tolist(),
                additivity_residual=float(endpoint-baseline-phi.sum()))

def explain(config):
    from functools import partial
    from threadpoolctl import threadpool_limits
    fit, outer, fit_keys, held=prepare(config)
    root=Path(config['project'])/'revision_outputs'
    old=root/config['fold_relative_directory']
    new=root/config['new_marker']
    require(digest(new)==config['marker_hash'],'Pilot kaydı değişmiş')
    def predictions(folder,record,name):
        require(digest(folder/name)==record['output_sha256'][name],'Tahmin dosyası değişmiş: '+name)
        df=pd.read_csv(folder/name,float_precision='round_trip')
        return df.loc[df.representation.eq('AE_latent21')]
    old_rows=predictions(old,json.loads((old/'fold_summary.json').read_text()),'outer_predictions.csv')
    new_rows=predictions(new.parent,json.loads(new.read_text()),'new_outer_predictions.csv.gz')
    saved=pd.concat([old_rows,new_rows],ignore_index=True)
    bg_idx=np.random.default_rng(42).choice(len(fit),16,replace=False)
    bg=fit[bg_idx]
    cases=[]
    results=[]
    for method in ['DBSCAN','IF','LOF','OC-SVM','kNN']:
        print('Doğrulama ve yerel açıklama:',method,flush=True)
        spec=config['selected'][method]
        threshold=spec['threshold']
        group=saved.loc[saved.method.eq(method)].set_index('sample_key',verify_integrity=True)
        require(set(group.index)==set(held.sample_key),'Dış örnek anahtarları uyuşmuyor')
        group=group.loc[held.sample_key]
        for col in ['UnitNumber','Time','RUL','late_life_le_30']:
            np.testing.assert_array_equal(group[col],held[col])
        if method in ['DBSCAN','OC-SVM']:
            model,_=fit_missing(method,spec['selected_params'],fit,{'max_dbscan_neighbor_entries':20000000})
            fn=partial(score_missing,method,model)
        else:
            model=fit_candidate(method,spec['selected_params'],fit,42)
            fn=partial(score,method,model)
        with threadpool_limits(limits=1):
            scores=fn(outer)
            np.testing.assert_allclose(scores,group.score,rtol=1e-5,atol=1e-7)
            np.testing.assert_allclose(group.threshold,threshold,rtol=0,atol=1e-12)
            np.testing.assert_array_equal(scores>=threshold,group.prediction)
            y=group.late_life_le_30.to_numpy()
            tags=np.where(y==1,np.where(scores>=threshold,'TP','FN'),np.where(scores>=threshold,'FP','TN'))
            for tag in ['TP','FP','FN','TN']:
                require(int((tags==tag).sum())==spec[tag],'Karar sayısı uyuşmuyor: '+method+tag)
            if method=='DBSCAN':
                for tag in ['TP','FP','FN','TN']:
                    indices=np.flatnonzero(tags==tag)
                    median=np.median(group.score.to_numpy()[indices])
                    idx=min(indices,key=lambda i:(abs(group.score.iloc[i]-median),str(group.index[i])))
                    cases.append((tag,int(idx)))
            for tag,idx in cases:
                item=contributions(fn,bg,outer[idx])
                np.testing.assert_allclose(item['score'],group.score.iloc[idx],rtol=1e-5,atol=1e-7)
                require((item['score']>=threshold)==bool(group.prediction.iloc[idx]),'Örnek karar eşiği uyuşmuyor')
                row=held.iloc[idx]
                item.update(method=method,anchor_category=tag,category=str(tags[idx]),sample_key=str(row.sample_key),
                            unit=int(row.UnitNumber),cycle=int(row.Time),RUL=int(row.RUL),prediction=int(group.prediction.iloc[idx]),
                            recorded_score=float(group.score.iloc[idx]),threshold=threshold,margin=item['score']-threshold,
                            z=outer[idx].tolist(),selected_params=spec['selected_params'])
                results.append(item)
    require(len(results)==20 and len(set(i for t,i in cases))==4,'Eksik karar açıklaması')
    return dict(status='PASS',dataset='FD001',fold=1,seed=42,representation='deep AE-21 standardized latent coordinates',
                background_keys=fit_keys.iloc[bg_idx].tolist(),background_size=16,permutation_pairs_per_background=8,
                reference='RUL <= 30',cases_rule='Nearest to median archived DBSCAN score in each TP/FP/FN/TN stratum, lexicographic sample key on ties.',
                outer_count=len(held),case_method_results=results)

if __name__=='__main__':
    config=json.loads(Path(sys.argv[1]).read_text())
    result=explain(config)
    Path(sys.argv[2]).write_text(json.dumps(result,allow_nan=False))
